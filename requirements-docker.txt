# do -- Docker's python:3.13-slim + build-essential (Dockerfile) can.

# two terms the same way hwcomponents-adc already replaced the ADC term.
hwcomponents-neurosim==1.0.43

# (hwcomponents_cacti.SRAM) -- ships a pre-built Linux binary (no compile

# at all on Windows; Docker's Linux image doesn't have that problem.
hwcomponents-cacti==1.0.40
